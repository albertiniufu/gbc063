\begin{frame}{Uma atualização, dois gargalos}
  \[
    \theta_{k+1} = \theta_k + \alpha\,\ghat_k,
    \qquad
    \ghat = \frac{1}{m}\sum_{i=1}^{m}\sum_{t}\gamma^t\,
      \nabla_\theta\log\pol_\theta\big(a_t^{(i)}\mid s_t^{(i)}\big)\,G_t^{(i)}
  \]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{teobox}{I · Estatística (Parte I)}
        \small Direção $\ghat_k$ não-viesada, mas ruidosa demais.\\
        \textbf{Rota:} linha de base $\to$ crítico $\to$ vantagem.
      \end{teobox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exbox}[title={II · Geometria (Parte II)}]{}
        \small Dados expiram a cada passo; $\|\Delta\theta\|$ não mede a mudança da política.\\
        \textbf{Rota:} KL/Fisher $\to$ substituto $\to$ PPO.
      \end{exbox}
    \end{column}
  \end{columns}

  \begin{alertabox}[title={Não-viesado não significa utilizável}]
    \[
      \mathrm{EQM}(\ghat) = \big\|\,\text{viés}\,\big\|^2 + \frac{\Var}{m}
    \]
  \end{alertabox}
\end{frame}

\section{Estimando a direção de subida}

\begin{frame}{A linha de base}
  \begin{defbox}{gradiente de política com linha de base}
    \[
      \nabla_\theta J(\theta) = \E_{\tau\sim\pol_\theta}\Big[\sum_{t}\gamma^t\,
        \nabla_\theta\log\pol_\theta(a_t\mid s_t)\,\big(G_t - b(s_t)\big)\Big]
    \]
  \end{defbox}
  \begin{itemize}\small
    \item Qualquer $b:\gS\to\R$ que dependa \alert{só do estado}: segue não-viesado.
    \item Leitura: reforçar $a_t$ na medida em que $G_t$ \textbf{superou} $b(s_t)$.
    \item Sem linha de base, ação medíocre em estado excelente é reforçada como excelente.
  \end{itemize}
\end{frame}

%% ============================================================
\begin{frame}{A linha de base não introduz viés}
  \footnotesize
  \begin{align*}
    &\E_{\tau}\big[\nabla_\theta\log\pol_\theta(a_t\mid s_t)\,b(s_t)\big]\\
    &\quad= \azul{\E_{s_{0:t},\,a_{0:t-1}}\Big[\,\E_{a_t,\,s_{t+1:\infty},\,a_{t+1:\infty}}}
       \big[\nabla_\theta\log\pol_\theta(a_t\mid s_t)\,b(s_t)\big]\azul{\Big]}
      && \text{\color{rlCinza}\itshape quebra da esperança}\\
    &\quad= \E\Big[\,\azul{b(s_t)\;\E_{a_t\sim\pol_\theta(\cdot\mid s_t)}}
       \big[\nabla_\theta\log\pol_\theta(a_t\mid s_t)\big]\Big]
      && \text{\color{rlCinza}\itshape $b$ sai}\\
    &\quad= \E\Big[\,b(s_t)\azul{\sum_{a}\pol_\theta(a\mid s_t)\,
       \frac{\nabla_\theta\pol_\theta(a\mid s_t)}{\pol_\theta(a\mid s_t)}}\Big]
       = \E\big[\,b(s_t)\,\azul{\nabla_\theta 1}\,\big] = \azul{0}
      && \text{\color{rlCinza}\itshape normalização}
  \end{align*}

  \begin{alertabox}
    Tudo se apoia em $\sum_a\pol_\theta(a\mid s) = 1$ para todo $\theta$.
    Com $b(s_t,a_t)$ (ex.: $b = \Qpi$) a fatoração quebra e o estimador fica \textbf{viesado}.
  \end{alertabox}
\end{frame}

%% ============================================================
\begin{frame}{Linha de base ótima}
  \begin{teobox}{Minimizando o segundo momento (mínimos quadrados ponderados)}
    \[
      b^*(s) = \frac{\E\big[\|\nabla_\theta\log\pol_\theta(a\mid s)\|^2\,G_t\mid s\big]}
                    {\E\big[\|\nabla_\theta\log\pol_\theta(a\mid s)\|^2\mid s\big]}
      \;\approx\;\E[G_t\mid s_t=s] = \Vpi(s)
    \]
  \end{teobox}
  \begin{itemize}\small
    \item Média dos retornos ponderada por $\|\nabla_\theta\log\pol_\theta\|^2$.
    \item Sem a ponderação (padrão na prática): $\Vpi(s)$.
          \alert{Linha de base prática = crítico de valor.}
  \end{itemize}
\end{frame}

  \framesubtitle{$\pol = (\tfrac13,\tfrac13,\tfrac13)$, $r = (98,101,104)$, $\ghat = (e_a-\pol)(r(a)-b)$}

      \midrule

      $b=b^*=101$ & $(-1,0,+1)$ & 4,00    & \azul{2,00} \\

  \end{center}

  \begin{ideiabox}
    Mesmo gradiente esperado; \textbf{$\approx$3400$\times$ menos variância}.
    O ganho é da ordem de (nível médio / dispersão)$^2$: 101 contra $\pm 3$.
  \end{ideiabox}
\end{frame}

%% ============================================================
\begin{frame}{Ator--crítico e vantagem}
  \begin{defbox}{métodos ator--crítico}
    \[
      \nabla_\theta J(\theta) \approx \E\Big[\sum_{t}\gamma^t\,
        \nabla_\theta\log\pol_\theta(a_t\mid s_t)\,\Ahat^{\pol}(s_t,a_t)\Big],
      \qquad
      A^{\pol}(s,a) \defeq \Qpi(s,a) - \Vpi(s)
    \]
  \end{defbox}
  \begin{itemize}\small
    \item Mesmo com $G_t - \Vpi$, $G_t$ acumula ruído até o fim do episódio: trocar MC por TD (\emph{bootstrapping}).
    \item Centralização: $\E_{a\sim\pol}[A^{\pol}(s,a)] = \Vpi(s)-\Vpi(s) = 0$.
    \item Ator--crítico = linha de base + \emph{bootstrapping}.
  \end{itemize}
\end{frame}

\begin{frame}{O espectro de estimadores}
  \footnotesize\setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Peso no estimador} & \textbf{Viés} & \textbf{Variância} \\
    \midrule
    $R(\tau)$ & Nenhum & Altíssima \\

  \end{tabular}

  \vspace{0.6em}
  \begin{alertabox}[title={Preço do TD}]
    $\E[\delta_t\mid s_t,a_t] = A^{\pol}(s_t,a_t)$ \textbf{apenas} quando $V_w = \Vpi$;
    caso contrário o viés vem do erro do crítico.
  \end{alertabox}
\end{frame}

\section{Geometria do passo e eficiência amostral}

\begin{frame}{As duas barreiras do gradiente \emph{on-policy}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{teobox}{1 · Ineficiência amostral}
        \small Após $\theta\leftarrow\theta+\alpha\ghat$, o lote descreve uma política que
        \textbf{não existe mais}.\\ \textbf{Meta do PPO:} vários passos por lote.
      \end{teobox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{teobox}{2 · Parâmetros vs. políticas}
        \small Passos em $\theta\in\Theta$; $J$ depende do deslocamento em $\Pi$.
        A régua $\|\Delta\theta\|$ não serve nos dois espaços.
      \end{teobox}
    \end{column}
  \end{columns}

  \begin{alertabox}[title={Colapso de desempenho}]
    Em RL, um passo grande demais destrói a política: ela passa a coletar só trajetórias ruins.
  \end{alertabox}
\end{frame}

%% ============================================================
\begin{frame}{Distância entre políticas: KL}
  \begin{defbox}{KL entre políticas no estado $s$}
    \[
      \KL(\pol'\,\|\,\pol)[s] \defeq \sum_{a}\pol'(a\mid s)\,\log\frac{\pol'(a\mid s)}{\pol(a\mid s)}
    \]
  \end{defbox}

  \begin{center}\small
    \begin{tabular}{@{}lrr@{}}
      \toprule
      \textbf{Sigmoide, $\Delta\theta = 2$} & $|\Delta\pol|$ & $\KL$ \\
      \midrule
      Região linear: $\theta: -1\to+1$ & \azul{0,4621} & \azul{0,4621} \\
      Região saturada: $\theta: 4\to6$ & {\color{rlLaranja}0,0155} & {\color{rlLaranja}0,0107} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{ideiabox}
    Mesmo $\Delta\theta$, comportamento 30$\times$ (probabilidade) e 43$\times$ (KL) diferente.
  \end{ideiabox}
\end{frame}

%% ============================================================
\begin{frame}{Geometria local: Fisher}
  \[
    d^{\pol}(s) \defeq (1-\gamma)\sum_{t}\gamma^t\,\Prob(s_t = s\mid\pol)
  \]
  \[
    \KLbar(\theta+\Delta\theta\,\|\,\theta) \defeq \E_{s\sim d^{\pol_\theta}}\big[\KL(\pol_{\theta+\Delta\theta}\,\|\,\pol_\theta)[s]\big]
      \approx\tfrac12\,\Delta\theta^\top F(\theta)\,\Delta\theta
  \]
  \[
    F(\theta) = \E_{s\sim d^{\pol_\theta},\,a\sim\pol_\theta}\big[\nabla_\theta\log\pol_\theta\,\nabla_\theta\log\pol_\theta^\top\big]
  \]
  \begin{itemize}\small
    \item \textbf{TRPO:} gradiente natural $F^{-1}g$ sob restrição de KL (2ª ordem).
    \item \textbf{PPO:} proximidade em $\Pi$ só com 1ª ordem, sem construir $F$.
  \end{itemize}
\end{frame}

%% ============================================================
\begin{frame}{Lema da diferença de desempenho}
  \begin{teobox}{Lema (Kakade \& Langford, 2002)}
    \[
      J(\pol') - J(\pol) = \frac{1}{1-\gamma}\,\E_{s\sim d^{\pol'},\,a\sim\pol'}\big[A^{\pol}(s,a)\big]
    \]
  \end{teobox}
  \begin{itemize}\small
    \item Prova: $A^{\pol}$ como resíduo de Bellman; a soma de $\gamma^{t+1}\Vpi(s_{t+1})-\gamma^t\Vpi(s_t)$ é telescópica.
    \item Leitura: $\pol'$ melhora se escolhe ações de vantagem positiva \textbf{nos estados que ela visita}.
  \end{itemize}
\end{frame}

%% ============================================================
\begin{frame}{O objetivo substituto}
  \begin{enumerate}\small
    \item \azul{$a\sim\pol'\to a\sim\pol$}: exato, amostragem por importância por passo.
    \item {\color{rlLaranja}$d^{\pol'}\to d^{\pol}$}: aproximação local de 1ª ordem.
  \end{enumerate}
  \[
    L_{\pol}(\pol') \defeq \E_{\tau\sim\pol}\Big[\sum_{t}\gamma^t\,\frac{\pol'(a_t\mid s_t)}{\pol(a_t\mid s_t)}\,A^{\pol}(s_t,a_t)\Big]
  \]
  \begin{ideiabox}
    Uma única razão por passo (não um produto ao longo da trajetória):
    estimável com o lote antigo, \textbf{vários passos de gradiente por coleta}.
  \end{ideiabox}
\end{frame}

%% ============================================================
\begin{frame}{Quando $L_{\pol}(\pol')$ é confiável?}
  \begin{teobox}{Teorema (Achiam et al., 2017)}
    \[
      \Big|\,J(\pol') - \big(J(\pol) + L_{\pol}(\pol')\big)\Big| \;\le\;
        C\,\sqrt{\E_{s\sim d^{\pol}}\big[\KL(\pol'\,\|\,\pol)[s]\big]}
    \]
  \end{teobox}
  \begin{itemize}\small
    \item Em $\pol'=\pol$: $L_{\pol}(\pol)=0$ e mesmo gradiente que $J$.
    \item Maximizar $J(\pol)+L_{\pol}(\pol')-C\sqrt{\KLbar}$ garante melhoria.
  \end{itemize}
  \begin{alertabox}
    Sem restrição, maximizar $L_{\pol}$ empurra $\pol'/\pol\gg1$ para fora da região de validade.
  \end{alertabox}
\end{frame}

%% ============================================================
\begin{frame}{PPO: duas maneiras de não se afastar demais}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{teobox}{Variante 1 · Penalidade KL adaptativa}
        \small
        \[
          \max_\theta\; L_{\theta_k}(\theta) - \beta_k\,\KLbar(\theta_k,\theta)
        \]
        $\KLbar\ge1{,}5\,\delta \Rightarrow \beta\leftarrow2\beta$;\quad
        $\KLbar\le\delta/1{,}5 \Rightarrow \beta\leftarrow\beta/2$.
      \end{teobox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exbox}[title={Variante 2 · Objetivo recortado}]{}
        \small
        \[
          r_t(\theta) = \frac{\pol_\theta(a_t\mid s_t)}{\pol_{\theta_k}(a_t\mid s_t)}
        \]
        \begin{align*}
          L^{\mathrm{CLIP}} = \E\Big[\sum_t\min\big(&r_t\Ahat_t,\\
            &\clipop(r_t,1{-}\epsilon,1{+}\epsilon)\,\Ahat_t\big)\Big]
        \end{align*}
      \end{exbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{PPO recortado na prática}
  \begin{algbox}{PPO-Clip}
    \small
    \textbf{para} $k = 0, 1, 2, \dots$
    \begin{enumerate}\setlength{\itemsep}{0pt}
      \item Colete $NT$ transições com $\pol_{\theta_k}$; calcule $\hat G_t$, $\Ahat_t$ (GAE) e \textbf{congele}.
      \item Guarde $\log\pol_{\theta_k}(a_t\mid s_t)$ como constante.
      \item $K\in[3,10]$ épocas de Adam em minilotes na perda:
    \end{enumerate}
    \[
      \mathcal{L}(\theta,w) = -L^{\mathrm{CLIP}} + c_1\big(V_w(s_t) - \hat G_t\big)^2 - c_2\,\mathcal{H}\big[\pol_\theta(\cdot\mid s_t)\big]
    \]
  \end{algbox}
  \begin{alertabox}[title={Erros fatais}]
    Recalcular $\Ahat_t$ dentro das épocas; derivar o denominador de $r_t$ (gradiente nulo);
    $K$ alto demais. \textbf{Monitore} a KL empírica e a fração com $|r_t-1|>\epsilon$.
  \end{alertabox}
\end{frame}

%% ============================================================
\begin{frame}{O que você deve saber ao sair daqui}
  \begin{enumerate}\small\setlength{\itemsep}{0.35em}
    \item \textbf{Demonstrar} que $b(s)$ não introduz viés e onde a prova quebra se $b$ depender da ação.
    \item \textbf{Explicar} por que a linha de base ótima é $\approx\Vpi(s)$ e escrever o gradiente com
          $A^{\pol} = \Qpi - \Vpi$.
    \item \textbf{Comparar} os estimadores de $R(\tau)$ a $\delta_t$ em viés e variância.
    \item \textbf{Distinguir} distância em $\Theta$ de distância em $\Pi$ (KL, Fisher).
    \item \textbf{Enunciar} o lema da diferença de desempenho e a aproximação que gera $L_{\pol}(\pol')$.
    \item \textbf{Escrever} $L^{\mathrm{CLIP}}$, desenhar seus ramos e justificar a assimetria.
  \end{enumerate}
\end{frame}
